3. True Atomic SQL Stock Update (`backend/app/repositories/product_repository.py`):
```python
@staticmethod
async def update_stock_atomic(db: AsyncSession, product_id: uuid.UUID, quantity: int) -> bool:
    stmt = (
        update(Product)
        .where(Product.id == product_id)
        .where(Product.stock >= quantity)
        .where(Product.is_deleted == False)
        .values(stock=Product.stock - quantity)
    )
    result = await db.execute(stmt)
    return bool(result.rowcount and result.rowcount > 0)
```

4. Request Payload Content Size Limit Middleware (`backend/app/main.py`):
```python
MAX_REQUEST_BODY_SIZE = 10 * 1024 * 1024  # 10MB

class ContentSizeLimitMiddleware(BaseHTTPMiddleware):
    async def dispatch(self, request: Request, call_next):
        content_length = request.headers.get("content-length")
        if content_length and int(content_length) > MAX_REQUEST_BODY_SIZE:
            return JSONResponse(
                status_code=status.HTTP_413_REQUEST_ENTITY_TOO_LARGE,
                content={"detail": "Request payload exceeds maximum allowed size (10MB)."},
            )
        return await call_next(request)
```
