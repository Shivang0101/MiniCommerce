1. Dual Engine Database Connection Splitting (`backend/app/db/session.py`):
```python
primary_engine = _create_engine(settings.DATABASE_URL)
replica_engine = _create_engine(settings.READ_DATABASE_URL) if settings.READ_DATABASE_URL else primary_engine

async def get_write_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionPrimary() as session:
        yield session

async def get_read_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionReplica() as session:
        try:
            yield session
        except Exception:
            async with AsyncSessionPrimary() as fallback_session:
                yield fallback_session
```
